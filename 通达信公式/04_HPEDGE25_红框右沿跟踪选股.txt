{公式名称:HPEDGE25}
{公式类型:条件选股公式}
{周期:日线}
{说明:v2.5右沿跟踪,仅选择前三日曾满足HPRED231但当前有限度右移的股票}

M5:=MA(C,5);
M10:=MA(C,10);
M15:=MA(C,15);
M30:=MA(C,30);

LOW60:=LLV(L,60);
LAGE:=LLVBARS(L,60);
LOWCORE:=LAGE>=5 AND LAGE<=35
 AND C<=LOW60*1.08
 AND C<=LLV(L,250)*1.15;

GAINCORE:=C<=REF(C,5)*1.05
 AND C<=REF(C,10)*1.06
 AND C<=REF(C,20)*1.07
 AND HHV(C/REF(C,1),10)<1.075;

MA5CORE:=C>=M5 OR (C>=M5*0.99 AND M5>=REF(M5,2)*0.995);
MA10CORE:=NOT(C>=M5 AND C>=M10)
 OR BARSLAST(C<M5 OR C<M10)<=5;
MACORE:=MA5CORE AND MA10CORE AND C<M30 AND C<=M15*1.01;

R10:=HHV(H,10)/LLV(L,10);
R20:=HHV(H,20)/LLV(L,20);
CR10:=HHV(C,10)/LLV(C,10);
CSLP:=SLOPE(C,10)/MA(C,10)*100;
FDRIFT:=LLV(L,5)/REF(LLV(L,10),5);

P1:=LAGE>=7;
P2:=R10<=1.125;
P3:=CR10<=1.08;
P4:=COUNT(L<=LOW60*1.04,15)>=3;
P5:=CSLP>=-0.30;
P6:=FDRIFT>=0.975;
PCNT:=IF(P1,1,0)+IF(P2,1,0)+IF(P3,1,0)
 +IF(P4,1,0)+IF(P5,1,0)+IF(P6,1,0);
PLAT:=PCNT>=5 AND R10<=1.14 AND CR10<=1.09;

L2:=LLV(L,8);
A2:=LLVBARS(L,8);
L1:=REF(LLV(L,25),8);
A1:=8+REF(LLVBARS(L,25),8);
SEP:=A1-A2;
DBL:=SEP>=5 AND SEP<=30
 AND L2>=L1*0.96 AND L2<=L1*1.05
 AND HHV(H,33)>=MAX(L1,L2)*1.035
 AND CR10<=1.10 AND CSLP>=-0.70;
STRUCT:=R10<=1.14 AND R20<=1.28 AND (PLAT OR DBL);

VOLCORE:=MA(V,5)<=MA(V,20)
 AND COUNT(V<=MA(V,20)*1.05,5)>=3;

LC:=REF(C,1);
RSI14:=SMA(MAX(C-LC,0),14,1)/SMA(ABS(C-LC),14,1)*100;
WRSI:="HPBASE.RSI14#WEEK";
WRSIUP:="HPBASE.RSIUP#WEEK";
RSICORE:=RSI14>=28 AND RSI14<=45 AND WRSI>=24 AND WRSI<=42;

DIF:=EMA(C,12)-EMA(C,26);
DEA:=EMA(DIF,9);
MBAR:=2*(DIF-DEA);
DUNDER:=DIF<0 AND DEA<0;
DNEW:=DUNDER AND COUNT(CROSS(DIF,DEA),6)>=1;
DIMP:=MBAR>REF(MBAR,3) AND DIF>=REF(DIF,3);
DMACD:=DUNDER AND (DNEW OR DIMP);

WDIF:="HPBASE.DIF#WEEK";
WDEA:="HPBASE.DEA#WEEK";
WIMP:="HPBASE.MACDIMP#WEEK";
WMACDCORE:=WDIF<0 AND WDEA<0 AND (WIMP>0 OR DNEW);

CAPOK:=FINANCE(41)>=5000000000
 AND FINANCE(40)>=3000000000;
LIQOK:=MA(AMOUNT,20)>=50000000;
PROFITOK:=FINANCE(30)>=10000000;
NAMEOK:=NOT(NAMELIKE('ST'))
 AND NOT(NAMELIKE('*'))
 AND NOT(NAMELIKE('退'));
HISTOK:=BARSCOUNT(C)>260;

CORE:=HISTOK AND NAMEOK AND PROFITOK
 AND CAPOK AND LIQOK
 AND LOWCORE AND GAINCORE AND MACORE
 AND STRUCT AND VOLCORE
 AND RSICORE AND DMACD AND WMACDCORE;

RECENTCORE:=COUNT(REF(CORE,1),3)>=1;

LOWEDGE:=LAGE>=5 AND LAGE<=38
 AND C<=LOW60*1.10
 AND C<=LLV(L,250)*1.15;
GAINEDGE:=C<=REF(C,5)*1.08
 AND C<=REF(C,10)*1.10
 AND C<=REF(C,20)*1.12
 AND HHV(C/REF(C,1),10)<1.075;
MAEDGE:=C>=M5*0.99
 AND C<=M15*1.03
 AND C<=M30*1.02;
RSIEDGE:=RSI14>=28 AND RSI14<=50
 AND WRSI>=24 AND WRSI<=45;
WMACDEDGE:=WDIF<0 AND WDEA<0;
DIREDGE:=RSI14>=REF(RSI14,3) OR WRSIUP>0 OR WIMP>0;

XG:HISTOK AND NAMEOK AND PROFITOK
 AND CAPOK AND LIQOK
 AND RECENTCORE AND NOT(CORE)
 AND LOWEDGE AND GAINEDGE AND MAEDGE
 AND STRUCT AND VOLCORE
 AND RSIEDGE AND DMACD AND WMACDEDGE AND DIREDGE;
